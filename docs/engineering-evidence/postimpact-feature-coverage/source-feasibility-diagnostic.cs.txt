    private static object Detail(FloridaContactMotion motion,FacilitySupportRegion region,FloridaBound time)
    {
        var evaluated=motion.Evaluate(time,out var q,out var dq);
        var domain=motion.ProveDomain(time,region,out var gap,out var slope);
        var up=FloridaVector.From(region.Up);var east=FloridaVector.From(region.East);var north=FloridaVector.From(region.North);
        var projection=FloridaVector.Dot(up,q);var k=up.NormSquared;FloridaBound radius=region.RadiusMetres;
        var eastCoordinate=FloridaVector.Dot(east,q);var northCoordinate=FloridaVector.Dot(north,q);
        var eastLeft=radius*eastCoordinate.Magnitude;var eastRight=region.InnerEastMetres*projection;
        var northLeft=radius*northCoordinate.Magnitude;var northRight=region.InnerNorthMetres*projection;
        var outerE=(FloridaBound)region.InnerEastMetres+region.BlendMetres;var outerN=(FloridaBound)region.InnerNorthMetres+region.BlendMetres;
        var guard=1-(outerE.Square()+outerN.Square())/radius.Square();var perpendicular=q-up*(projection/k);
        var capLeft=projection.Square()*(k-guard.Square());var capRight=guard.Square()*k*perpendicular.NormSquared;
        var clampLeft=radius+region.PlaneAltitudeMetres;var clampRight=radius*k.Sqrt();
        var centerTime=time.Lower+(time.Upper-time.Lower)*.5;FloridaBound t=centerTime;
        var seed=(CartesianState)Field(motion,"seed");var linear=(NovaCore.Simulation.Spacecraft.Translation.SpacecraftTranslationState)Field(motion,"linear");
        var offset=(FloridaVector)Field(motion,"offset");var acceleration=(FloridaVector)Field(motion,"craftAcceleration");
        var earthA=(FloridaVector)Field(motion,"earthAcceleration");var earthJ=(FloridaVector)Field(motion,"earthJerk");
        var snap=(FloridaBound)Field(motion,"earthSnap");var dt=t-FloridaBound.Integer(linear.Epoch.Ticks)/SimulationInstant.TicksPerSecond;
        var relative=FloridaVector.From(linear.PositionRoot)-FloridaVector.From(seed.Position)+offset+FloridaVector.From(linear.VelocityRoot)*dt+
            acceleration*(dt.Square()/2)-FloridaVector.From(seed.Velocity)*t-earthA*(t.Square()/2)-earthJ*(t*t*t/6);
        var earthRemainder=snap*FloridaBound.Pow(FloridaBound.Point(Math.Abs(centerTime)),4)/24;
        return new{time,evaluated,domain,gap,slope,q,dq,projection,eastCoordinate,northCoordinate,
            guards=new{positiveProjection=projection.Lower>0,eastLeft,eastRight,eastPass=eastLeft.Upper<=eastRight.Lower,
                northLeft,northRight,northPass=northLeft.Upper<=northRight.Lower,guard,capLeft,capRight,capPass=capLeft.Lower>=capRight.Upper,
                clampLeft,clampRight,clampPass=clampLeft.Lower>clampRight.Upper},
            center=new{centerTime,relativeBeforeEarthRemainder=relative,earthRemainder,offset}};
    }
