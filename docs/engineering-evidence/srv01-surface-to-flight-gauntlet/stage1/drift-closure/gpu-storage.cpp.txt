#include <vulkan/vulkan.h>

int main(int argc,char**argv){
 VkInstanceCreateInfo ic{VK_STRUCTURE_TYPE_INSTANCE_CREATE_INFO};VkInstance in{};

  printf("GPU_BUFFER_REQUIREMENTS device=%s buffers=%d payload=%llu required=%llu driverInternalExcluded=true\n",prop.deviceName,argc-1,requested,allocated);vkDestroyDevice(d,nullptr);
 }vkDestroyInstance(in,nullptr);return 0;
}
