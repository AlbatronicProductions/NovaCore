static double Minimum(AssemblyContactProfile p,AssemblyMotion m)
{
    var min=double.MaxValue;var q=m.BodyToWorld;
    var a=2*(q.X*q.Y+q.Z*q.W);var b=1-2*(q.X*q.X+q.Z*q.Z);var c=2*(q.Y*q.Z-q.X*q.W);
    foreach(var ch in p.Children)for(var k=0;k<8;k++)
    {
        var x=((k&1)==0?-.5:.5)*ch.Dimensions.X;var y=((k&2)==0?-.5:.5)*ch.Dimensions.Y;var z=((k&4)==0?-.5:.5)*ch.Dimensions.Z;var r=ch.AtOrigin.Rotation;var o=ch.AtOrigin.Position;
        min=Math.Min(min,m.PositionO.Y+a*(o.X+r.A*x+r.B*y+r.C*z)+b*(o.Y+r.D*x+r.E*y+r.F*z)+c*(o.Z+r.G*x+r.H*y+r.I*z)+1.7);
    }
    return min;
}
